"use strict";
const assert=require("node:assert/strict"),{load}=require("/home/somanos/github/infra-layer/test/network-attachments/load.cjs");
const results=[];
function fixture(){
 const project="drumee-independent-final",id="a".repeat(64),ep="b".repeat(64),networks={};
 const composition={project,compose:{services:{redis:{networks:["kernel"]}},networks:{kernel:{name:project+"-kernel"},front:{name:project+"-front"}}}};
 for(const [i,role] of ["kernel","front"].entries())networks[role]={id:String(i+1).repeat(64),name:composition.compose.networks[role].name,owner:project,project,driver:"bridge",internal:false,containers:{}};
 const state={composition,networks,id,ep,calls:[],inventory:[id],failure:false,resource:{id,name:"/"+project+"-redis-1",owner:project,project,service:"redis",running:true,networks:{}}};
 state.attach=role=>{const n=networks[role];const endpoint=role==="kernel"?ep:"d".repeat(64);n.containers[id]={Name:state.resource.name.slice(1),EndpointID:endpoint,MacAddress:"02:42:ac:12:00:02",IPv4Address:"172.18.0.2/16",IPv6Address:""};state.resource.networks[n.name]={NetworkID:n.id,EndpointID:endpoint};};
 state.attach("kernel");
 state.driver=load((program,args)=>{assert.equal(program,"/usr/bin/docker");assert.equal(args[0],"--host");assert.equal(args[1],"unix:///var/run/docker.sock");args=args.slice(2);state.calls.push(args);if(state.failure&&args[0]==="ps")throw Object.assign(Error("independent-command-failure"),{status:125});if(args[0]==="ps"){assert.ok(args.includes("-a")&&args.includes("--no-trunc")&&args.includes("label=com.docker.compose.project="+project));return state.inventory.join("\n");}if(args[0]==="network"&&args[1]==="ls")return Object.values(networks).map(n=>n.name).join("\n");if(args[0]==="network"&&args[1]==="inspect")return JSON.stringify(Object.values(networks).find(n=>n.name===args.at(-1)));if(args[0]==="container"&&args[1]==="inspect"){assert.equal(args.at(-1),id);return JSON.stringify(state.resource);}throw Error("UNAPPROVED_COMMAND");});
 state.check=()=>state.driver.checkNetworks(composition,{require_existing:true});return state;
}
function control(name,fn){const f=fixture();fn(f);assert.ok(f.calls.every(args=>["network","container","ps"].includes(args[0])));results.push(name);}
const refusal=(f,code="NETWORK_ATTACHMENT_MISMATCH")=>assert.throws(f.check,{code});
control("nominal exact membership",f=>assert.equal(f.check(),undefined));
control("original R1 undeclared extra membership refused",f=>{f.resource.networks.foreign={NetworkID:"e".repeat(64),EndpointID:"f".repeat(64)};refusal(f);});
control("matching owner cannot authorize additional attachment",f=>{f.resource.networks.foreign={NetworkID:f.networks.front.id,EndpointID:"f".repeat(64),owner:f.composition.project};refusal(f);});
control("missing mandatory membership refused",f=>{f.resource.networks={};f.networks.kernel.containers={};refusal(f);});
control("disconnected stopped container still discovered",f=>{f.resource.running=false;f.resource.networks={};f.networks.kernel.containers={};refusal(f);assert.ok(f.calls.some(args=>args[0]==="container"));});
control("valid declared multihoming accepted",f=>{f.composition.compose.services.redis.networks={kernel:{},front:{}};f.attach("front");assert.equal(f.check(),undefined);});
control("wrong network ID refused",f=>{f.resource.networks[f.networks.kernel.name].NetworkID=f.networks.front.id;refusal(f);});
control("wrong endpoint identity refused",f=>{f.resource.networks[f.networks.kernel.name].EndpointID="f".repeat(64);refusal(f);});
control("foreign ownership refused",f=>{f.resource.owner="foreign";refusal(f,"NETWORK_OWNERSHIP_MISMATCH");});
control("container name identity substitution refused",f=>{f.resource.name="/foreign-redis-1";refusal(f,"INVALID_NETWORK_ATTACHMENT");});
control("Docker inventory failure stays fatal",f=>{f.failure=true;assert.throws(f.check,error=>error.code==="DOCKER_LIFECYCLE_COMMAND_FAILED"&&error.exit_status===125);});
control("synthetic pending endpoint never grants readiness",f=>{const meta=f.networks.kernel.containers[f.id];f.networks.kernel.containers={["ep-"+f.ep]:meta};refusal(f,"NETWORK_ATTACHMENT_UNSETTLED");});
control("synthetic endpoint cannot mask undeclared membership",f=>{const meta=f.networks.kernel.containers[f.id];f.networks.kernel.containers={["ep-"+f.ep]:meta};f.resource.networks.extra={NetworkID:"e".repeat(64),EndpointID:"f".repeat(64)};refusal(f);});
console.log(JSON.stringify({independent_controls:results.length,results,actual_fixed_module:true,Docker_commands_executed:0}));
